/**
 * Capas de profundidad para glassmorphism
 */
@Immutable
data class DepthLayer(
    val elevation: Int,
    val blur: Float,
    val opacity: Float,
    val borderOpacity: Float
) {
    companion object {
        val BASE = DepthLayer(0, 0f, 1f, 0f)
        val SURFACE = DepthLayer(1, 8f, 0.9f, 0.08f)
        val ELEVATED = DepthLayer(3, 16f, 0.95f, 0.12f)
        val OVERLAY = DepthLayer(5, 24f, 0.98f, 0.16f)
        val MODAL = DepthLayer(8, 32f, 1f, 0.2f)
    }
}

/**
 * Paleta completa con soporte para gradientes y glassmorphism
 */
@Immutable
data class NebulaPalette(
    // Fondos base
    val background: Color,
    val backgroundDeep: Color,
    val backgroundGradient: Brush,

    // Superficies con glassmorphism
    val surface: Color,
    val surfaceVariant: Color,
    val surfaceElevated: Color,
    val surfaceGlass: Color,      // Semi-transparente para glassmorphism
    val surfaceGlassStrong: Color,

    // Colores de acento (dinámicos por CosmicAccent)
    val primary: Color,
    val primaryGradient: Brush,
    val primaryGlow: Color,
    val onPrimary: Color,
    val surfaceTint: Color,

    // Estados
    val accent: Color,
    val warning: Color,
    val error: Color,
    val success: Color,

    // Texto
    val onBackground: Color,
    val onSurface: Color,
    val onSurfaceVariant: Color,
    val muted: Color,
    val outline: Color,

    // Capas de profundidad
    val depthSurface: DepthLayer,
    val depthElevated: DepthLayer,
    val depthOverlay: DepthLayer,
    val depthModal: DepthLayer,

    // Radios de esquina (más orgánicos)
    val radiusSmall: Float = 8f,
    val radiusMedium: Float = 12f,
    val radiusLarge: Float = 16f,
    val radiusFull: Float = 999f,

    // Sombras suaves
    val shadowColor: Color,
    val shadowOpacity: Float = 0.15f
)

/**
 * Tema principal de la app - Nebula
 */
@Composable
fun NebulaTheme(
    themeMode: ThemeMode = ThemeMode.DARK,
    cosmicAccent: CosmicAccent = CosmicAccent.NEBULA_VIOLET,
    content: @Composable () -> Unit
) {
    val basePalette = if (themeMode == ThemeMode.DARK) DarkNebula else LightNebula
    val palette = basePalette.copy(
        primary = cosmicAccent.primary,
        primaryGradient = cosmicAccent.gradient,
        primaryGlow = cosmicAccent.glow,
        onPrimary = cosmicAccent.onPrimary,
        surfaceTint = cosmicAccent.surfaceTint
    )

    // ColorScheme Material3 para compatibilidad
    val colorScheme = if (themeMode == ThemeMode.DARK) {
        darkColorScheme(
            primary = palette.primary,
            onPrimary = palette.onPrimary,
            primaryContainer = palette.surfaceTint.copy(alpha = 0.2f),
            onPrimaryContainer = palette.onBackground,
            secondary = palette.accent,
            onSecondary = palette.onBackground,
            secondaryContainer = palette.surfaceVariant,
            onSecondaryContainer = palette.onSurface,
            tertiary = palette.warning,
            onTertiary = palette.onBackground,
            error = palette.error,
            onError = Color.White,
            errorContainer = palette.error.copy(alpha = 0.15f),
            onErrorContainer = palette.error,
            background = palette.background,
            onBackground = palette.onBackground,
            surface = palette.surface,
            onSurface = palette.onSurface,
            surfaceVariant = palette.surfaceVariant,
            onSurfaceVariant = palette.onSurfaceVariant,
            outline = palette.outline,
            outlineVariant = palette.outline.copy(alpha = 0.5f),
            shadow = palette.shadowColor,
            scrim = palette.shadowColor,
            inverseSurface = palette.onBackground,
            inverseOnSurface = palette.background,
            inversePrimary = palette.primaryGlow
        )
    } else {
        lightColorScheme(
            primary = palette.primary,
            onPrimary = palette.onPrimary,
            primaryContainer = palette.surfaceTint.copy(alpha = 0.15f),
            onPrimaryContainer = palette.onBackground,
            secondary = palette.accent,
            onSecondary = palette.onBackground,
            secondaryContainer = palette.surfaceVariant,
            onSecondaryContainer = palette.onSurface,
            tertiary = palette.warning,
            onTertiary = palette.onBackground,
            error = palette.error,
            onError = Color.White,
            errorContainer = palette.error.copy(alpha = 0.1f),
            onErrorContainer = palette.error,
            background = palette.background,
            onBackground = palette.onBackground,
            surface = palette.surface,
            onSurface = palette.onSurface,
            surfaceVariant = palette.surfaceVariant,
            onSurfaceVariant = palette.onSurfaceVariant,
            outline = palette.outline,
            outlineVariant = palette.outline.copy(alpha = 0.5f),
            shadow = palette.shadowColor,
            scrim = palette.shadowColor,
            inverseSurface = palette.onBackground,
            inverseOnSurface = palette.background,
            inversePrimary = palette.primary
        )
    }

    CompositionLocalProvider(
        LocalNebulaPalette provides palette,
        LocalCosmicAccent provides cosmicAccent,
        LocalThemeMode provides themeMode
    ) {
        MaterialTheme(
            colorScheme = colorScheme,
            typography = NebulaTypography,
            shapes = NebulaShapes,
            content = content
        )
    }
}

/**
 * Helper para crear gradiente personalizado de servidor/usuario
 */
@Composable
fun rememberServerGradient(
    serverId: String,
    accent: CosmicAccent = CosmicAccent.NEBULA_VIOLET
): Brush {
    return remember(serverId, accent) {
        // Generar gradiente determinista basado en ID
        val hash = serverId.hashCode()
        val hueOffset = (hash % 360).toFloat()

        // Usar el acento base pero con variación
        Brush.sweepGradient(
            center = androidx.compose.ui.geometry.Offset(0.5f, 0.5f),
            colors = listOf(
                accent.primary.copy(alpha = 0.8f),
                accent.glow.copy(alpha = 0.6f),
                accent.primary.copy(alpha = 0.4f),
                accent.glow.copy(alpha = 0.2f),
                accent.primary.copy(alpha = 0.8f)
            ),
            startAngle = hueOffset,
            endAngle = hueOffset + 360f
        )
    }
}

/**
 * Helper para gradiente de avatar (aura)
 */
@Composable
fun rememberAvatarAura(
    userId: String,
    accent: CosmicAccent = CosmicAccent.NEBULA_VIOLET
): Brush {
    return remember(userId, accent) {
        val hash = userId.hashCode()
        val baseHue = (hash % 360).toFloat()

        Brush.radialGradient(
            center = androidx.compose.ui.geometry.Offset(0.5f, 0.5f),
            radius = 0.8f,
            colors = listOf(
                accent.primary.copy(alpha = 0.0f),
                accent.primary.copy(alpha = 0.15f),
                accent.glow.copy(alpha = 0.35f),
                accent.primary.copy(alpha = 0.5f)
            )
        )
    }
}
